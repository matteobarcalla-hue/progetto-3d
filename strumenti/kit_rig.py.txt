# --------------------------------------------------------------- persone e animali articolati (forme in MODELLI)
COLORI_BASE = {'fur_white': (0.92, 0.92, 0.9), 'fur_brown': (0.45, 0.28, 0.16), 'fur_black': (0.12, 0.11, 0.1), 'fur_tan': (0.72, 0.55, 0.36),
               'fur_grey': (0.52, 0.5, 0.48), 'feather_white': (0.95, 0.95, 0.93), 'wool_white': (0.9, 0.88, 0.82), 'beak_orange': (0.9, 0.55, 0.12),
               'stone_dark': (0.36, 0.35, 0.33), 'stone_light': (0.72, 0.69, 0.62), 'trunk_brown': (0.35, 0.24, 0.14), 'wall_plaster_rose': (0.86, 0.66, 0.6),
               'flag_red': (0.7, 0.12, 0.1), 'hedge': (0.18, 0.36, 0.16), 'wood_trim': (0.36, 0.24, 0.14), 'iron_dark': (0.25, 0.25, 0.27), 'hay': (0.8, 0.7, 0.4)}
_MESH = {}

def mat_mappa(nome):
    m = bpy.data.materials.get(nome) or next((x for x in bpy.data.materials if x.name.split('.')[0] == nome), None)
    return m or materiale(nome, COLORI_BASE.get(nome, (0.6, 0.6, 0.6)))

def mesh_parte(chiave, parte, mappa):
    """mesh condivisa fra tutti gli esemplari con la stessa forma e gli stessi colori"""
    k = (chiave, parte['nome'], tuple(sorted(mappa.items())))
    me = _MESH.get(k)
    if me is not None:
        return me
    me = bpy.data.meshes.new('npc_%s_%s' % (chiave, parte['nome']))
    me.from_pydata(parte['V'], [], parte['F'])
    nomi = []
    for t in parte['M']:
        n = mappa.get(t, t)
        if n not in nomi:
            nomi.append(n)
    for n in nomi:
        me.materials.append(mat_mappa(n))
    me.polygons.foreach_set('material_index', [nomi.index(mappa.get(t, t)) for t in parte['M']])
    me.update(); me['kit'] = True
    _MESH[k] = me
    return me

def rig(nome, chiave, mappa, coll):
    root = bpy.data.objects.new(nome, None); coll.objects.link(root); root.empty_display_size = 0.3
    obs = {}
    for p in MODELLI[chiave]:
        ob = bpy.data.objects.new(nome + '_' + p['nome'], mesh_parte(chiave, p, mappa)); coll.objects.link(ob)
        ob.parent = obs[p['genitore']] if p['genitore'] else root
        ob.location = p['perno']
        obs[p['nome']] = ob
    root['passo'] = 1.0
    return root, obs

def drv(ob, asse, espr, prop='rotation_euler', passo=None):
    fc = ob.driver_add(prop, asse); d = fc.driver; d.type = 'SCRIPTED'
    if passo is not None:
        v = d.variables.new(); v.name = 'p'; v.type = 'SINGLE_PROP'
        v.targets[0].id = passo; v.targets[0].data_path = '["passo"]'
    d.expression = espr
    return fc

def lunghezza(pts, chiusa=True):
    L = sum((b - a).length for a, b in zip(pts, pts[1:]))
    return L + ((pts[0] - pts[-1]).length if chiusa else 0)

def _percorso(root, cur):
    c = root.constraints.new('FOLLOW_PATH'); c.target = cur; c.use_fixed_location = True
    c.use_curve_follow = True; c.forward_axis = 'FORWARD_Y'; c.up_axis = 'UP_Z'
    # in salita e in discesa il corpo resta dritto: tengo solo la rotazione attorno alla verticale
    l = root.constraints.new('LIMIT_ROTATION'); l.owner_space = 'WORLD'
    l.use_limit_x = True; l.min_x = l.max_x = 0.0; l.use_limit_y = True; l.min_y = l.max_y = 0.0
    return c

def _chiavi(c, pts):
    """pts: [(fotogramma, posizione non avvolta)] -> chiavi di offset_factor con il giro 1 -> 0"""
    for (f0, p0), (f1, p1) in zip(pts, pts[1:]):
        c.offset_factor = p0 % 1.0; c.keyframe_insert('offset_factor', frame=f0)
        k = math.floor(p0) + 1
        while k <= p1 and p1 > p0:
            fk = f0 + (k - p0) / (p1 - p0) * (f1 - f0)
            c.offset_factor = 1.0; c.keyframe_insert('offset_factor', frame=fk - 0.01)
            c.offset_factor = 0.0; c.keyframe_insert('offset_factor', frame=fk)
            k += 1
    f1, p1 = pts[-1]; c.offset_factor = p1 % 1.0; c.keyframe_insert('offset_factor', frame=f1)

def cammina(root, cur, L, vel, fase, ftot):
    """cammino continuo (persone, uccelli in volo)"""
    c = _percorso(root, cur)
    P = max(30.0, L / max(vel, 0.05) * FPS); f0 = -P
    _chiavi(c, [(f0, fase), (ftot + P, fase + (ftot + 2 * P) / P)])
    interp(root, 'LINEAR')

def pascolo(root, cur, L, vel, fase, ftot, rnd, muovi=(4.0, 9.0), ferma=(3.0, 9.0)):
    """tratti di cammino alternati a soste: durante le soste le zampe si fermano (proprietà 'passo') e l'animale bruca o si guarda intorno"""
    c = _percorso(root, cur)
    ferma = (max(0.2, ferma[0] * SOSTE_ANIMALI), max(0.3, ferma[1] * SOSTE_ANIMALI))
    f = -rnd.uniform(0, 200); pos = fase; pts = [(f, pos)]; pk = []
    fermo = rnd.random() < 0.5; inizio = 0.0 if fermo else 1.0
    while f < ftot + 50:
        if fermo:
            f += rnd.uniform(*ferma) * FPS; pts.append((f, pos)); pk += [(f - 5, 0.0), (f + 5, 1.0)]
        else:
            T = rnd.uniform(*muovi) * FPS; pos += vel * T / FPS / max(L, 0.5); f += T; pts.append((f, pos)); pk += [(f - 5, 1.0), (f + 5, 0.0)]
        fermo = not fermo
    _chiavi(c, pts)
    root['passo'] = inizio; root.keyframe_insert('["passo"]', frame=pts[0][0])
    for fr, val in pk:
        root['passo'] = val; root.keyframe_insert('["passo"]', frame=fr)
    interp(root, 'LINEAR')

# parametri di andatura: velocità (m/s), lunghezza gamba (m), ampiezza (rad), bruca
ANDATURA = {'cow': (0.6, 0.93, 0.30, True), 'horse': (0.9, 1.17, 0.32, True), 'donkey': (0.6, 0.83, 0.32, True), 'sheep': (0.45, 0.51, 0.35, True),
            'goat': (0.55, 0.54, 0.35, True), 'pig': (0.45, 0.46, 0.35, False), 'deer': (0.75, 0.81, 0.32, True), 'dog': (1.0, 0.36, 0.40, False)}
ZAMPE4 = (('post_sx', 0.0), ('ant_sx', 0.5 * math.pi), ('post_dx', math.pi), ('ant_dx', 1.5 * math.pi))

def _w(v, l, A):
    return 2 * math.pi * v / (FPS * 4 * l * math.sin(A))

def anima_quadrupede(root, obs, sp, fase):
    v, l, A, bruca = ANDATURA[sp]; v *= VELOCITA_ANIMALI
    ph = 'frame*%.4f+%.3f' % (_w(v, l, A), fase)
    c = obs['corpo']; z0 = c.location.z
    drv(c, 2, '%.3f+p*0.018*abs(sin(%s))' % (z0, ph), prop='location', passo=root)
    drv(c, 1, 'p*0.025*sin(%s)' % ph, passo=root)
    k = obs['collo']
    if bruca:
        drv(k, 0, 'p*0.05*sin(2*(%s))-(1-p)*(0.45+0.35*max(0,sin(frame*0.013+%.3f)))' % (ph, fase * 1.7), passo=root)
    else:
        drv(k, 0, 'p*0.05*sin(2*(%s))-(1-p)*0.2*max(0,sin(frame*0.02+%.3f))' % (ph, fase), passo=root)
    drv(k, 2, '0.3*sin(frame*0.011+%.3f)*(1-0.6*p)' % (fase * 2.3), passo=root)
    if sp == 'dog':
        drv(obs['coda'], 2, '0.5*sin(frame*0.55+%.3f)' % fase)
    else:
        drv(obs['coda'], 1, '0.3*sin(frame*0.09+%.3f)+0.1*sin(frame*0.23+%.3f)' % (fase, fase * 1.3))
    for nome, o in ZAMPE4:
        drv(obs['zampa_' + nome], 0, 'p*%.3f*sin(%s+%.3f)' % (A, ph, o), passo=root)
        drv(obs['stinco_' + nome], 0, '-p*%.3f*max(0,cos(%s+%.3f))' % (A * 2.2, ph, o), passo=root)
    return v

def anima_gallina(root, obs, fase):
    vg = 0.35 * VELOCITA_ANIMALI
    ph = 'frame*%.4f+%.3f' % (_w(vg, 0.2, 0.5), fase)
    c = obs['corpo']
    drv(c, 2, '%.3f+p*0.012*abs(sin(%s))' % (c.location.z, ph), prop='location', passo=root)
    drv(obs['collo'], 0, 'p*0.18*sin(2*(%s))-(1-p)*1.0*max(0,sin(frame*0.22+%.3f)-0.2)' % (ph, fase), passo=root)
    drv(obs['collo'], 2, '(1-p)*0.4*sin(frame*0.05+%.3f)' % fase, passo=root)
    for s, o in (('sx', 0.0), ('dx', math.pi)):
        drv(obs['zampa_' + s], 0, 'p*0.5*sin(%s+%.3f)' % (ph, o), passo=root)
    return vg

def anima_acqua(root, obs, fase):
    c = obs['corpo']
    drv(c, 2, '%.3f+0.012*sin(frame*0.09+%.3f)' % (c.location.z, fase), prop='location')
    drv(c, 1, '0.03*sin(frame*0.07+%.3f)' % fase)
    drv(obs['collo'], 2, '0.35*sin(frame*0.017+%.3f)' % fase)
    drv(obs['collo'], 0, '-0.9*max(0,sin(frame*0.012+%.3f)-0.7)' % (fase * 1.9))
    return 0.22 * VELOCITA_ANIMALI

def anima_gabbiano(root, obs, fase, vola):
    c = obs['corpo']
    if vola:
        drv(obs['ala_sx'], 1, '0.55*sin(frame*0.32+%.3f)' % fase)
        drv(obs['ala_dx'], 1, '-0.55*sin(frame*0.32+%.3f)' % fase)
        drv(c, 2, '%.3f+0.08*sin(frame*0.32+%.3f)' % (c.location.z, fase + 1.2), prop='location')
        return 9.0
    vg = 0.4 * VELOCITA_ANIMALI
    ph = 'frame*%.4f+%.3f' % (_w(vg, 0.11, 0.5), fase)
    drv(obs['collo'], 2, '0.5*sin(frame*0.03+%.3f)' % fase)
    for s, o in (('sx', 0.0), ('dx', math.pi)):
        drv(obs['zampa_' + s], 0, 'p*0.5*sin(%s+%.3f)' % (ph, o), passo=root)
    return vg

def anima_persona(root, obs, v, fase, guardia=False):
    A = 0.42; ph = 'frame*%.4f+%.3f' % (_w(v, 0.9, A), fase)
    b = obs['bacino']
    drv(b, 2, '%.3f+0.03*abs(sin(%s))' % (b.location.z, ph), prop='location')
    drv(b, 2, '0.07*sin(%s)' % ph)
    drv(obs['collo'], 2, '0.3*sin(frame*0.017+%.3f)' % (fase * 1.3))
    for s, o in (('sx', 0.0), ('dx', math.pi)):
        drv(obs['anca_' + s], 0, '%.3f*sin(%s+%.3f)' % (A, ph, o))
        drv(obs['ginocchio_' + s], 0, '-0.9*max(0,cos(%s+%.3f))' % (ph, o))
        if guardia and s == 'dx':
            obs['spalla_dx'].rotation_euler = (0.1, 0, 0); obs['gomito_dx'].rotation_euler = (-0.1, 0, 0)
            continue
        drv(obs['spalla_' + s], 0, '-%.3f*sin(%s+%.3f)' % (0.12 if guardia else 0.35, ph, o))
        drv(obs['gomito_' + s], 0, '0.25+0.2*max(0,sin(%s+%.3f))' % (ph, o))
